function drawConnections() { const svg = document.getElementById('connections-layer'); svg.innerHTML = ''; const containerRect = document.querySelector('.board-container').getBoundingClientRect(); state.connections.forEach((conn, index) => { const fromCard = document.getElementById(`card-${conn.from}`); const toCard = document.getElementById(`card-${conn.to}`); if (!fromCard || !toCard) return; const subjFrom = state.subjects.find(s => s.id === conn.from); const subjTo = state.subjects.find(s => s.id === conn.to); const rectFrom = fromCard.getBoundingClientRect(); const rectTo = toCard.getBoundingClientRect(); const isSameLevel = (subjFrom && subjTo && subjFrom.level === subjTo.level); // --- EL TRUCO PARA SEPARAR LAS LÍNEAS VERTICALES --- // Generamos un desfase en X (izquierda a derecha) basado en el número de línea. // Valores posibles: -20px, -10px, 0, +10px, +20px respecto al centro. const xOffset1 = (index % 5) * 10 - 20; const xOffset2 = ((index + 2) % 5) * 10 - 20; // Un cálculo diferente para que no sean idénticos al llegar const x1 = rectFrom.left + rectFrom.width / 2 - containerRect.left + xOffset1; const x2 = rectTo.left + rectTo.width / 2 - containerRect.left + xOffset2; // --------------------------------------------------- let pathData = ""; if (isSameLevel) { // Correquisitos const y1 = rectFrom.top - containerRect.top; const y2 = rectTo.top - containerRect.top; const upOffset = 15 + (index % 4) * 6; pathData = `M ${x1} ${y1} V ${y1 - upOffset} H ${x2} V ${y2}`; } else { // Prerrequisitos const y1 = rectFrom.bottom - containerRect.top; const y2 = rectTo.top - containerRect.top; const midY = y1 + (y2 - y1) / 2; const offset = (index % 7) * 5 - 15; pathData = `M ${x1} ${y1} V ${midY + offset} H ${x2} V ${y2}`; } const path = document.createElementNS('http://www.w3.org/2000/svg', 'path'); path.setAttribute('d', pathData); path.setAttribute('class', `connection ${isSameLevel ? 'corequisite' : ''}`); path.style.strokeLinejoin = "round"; path.style.strokeLinecap = "round"; path.dataset.from = conn.from; path.dataset.to = conn.to; svg.appendChild(path); }); if (pinnedSubjectId) { highlightConnections(pinnedSubjectId); } }
